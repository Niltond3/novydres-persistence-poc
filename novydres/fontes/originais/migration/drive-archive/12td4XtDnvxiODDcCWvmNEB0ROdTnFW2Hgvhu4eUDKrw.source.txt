import copy
import unittest
from novydres_stage3_prototype import *
class TestStage3(unittest.TestCase):
 def setUp(self):self.a=new_sandbox()
 def add(self):return append(self.a,event_id='TEST-EVT-001',actor='system',kind='SYSTEM_TEST',text='Synthetic test only',update={'flag':'ONE'})[0]
 def test_01_new_valid(self):self.assertTrue(verify(self.a))
 def test_02_append_and_state(self):
  x=self.add();self.assertEqual(x['checkpoint']['flag'],'ONE');self.assertEqual(x['last_seq'],1)
 def test_03_duplicate_id_idempotent(self):
  x=self.add();y,status=append(x,event_id='TEST-EVT-001',actor='system',kind='SYSTEM_TEST',text='Synthetic test only',update={'flag':'ONE'});self.assertEqual(status,'ALREADY_COMMITTED');self.assertEqual(x,y)
 def test_04_same_id_conflicting_payload_rejected(self):
  x=self.add();self.assertRaises(Rejected,append,x,event_id='TEST-EVT-001',actor='system',kind='SYSTEM_TEST',text='Different')
 def test_05_tampered_event_hash(self):
  x=self.add();x['events'][0]['text']='falsified';self.assertRaises(Rejected,verify,x)
 def test_06_tampered_checkpoint_missing(self):
  x=self.add();del x['checkpoint'];self.assertRaises(Rejected,verify,x)
 def test_07_no_pc_action_without_decl(self):
  self.assertRaises(Rejected,append,self.a,event_id='TEST-EVT-01',actor='pc',kind='PLAYER_DECLARATION',text='I move')
 def test_08_npc_claim_cannot_set_state(self):
  self.assertRaises(Rejected,append,self.a,event_id='TEST-EVT-01',actor='npc',kind='NPC_CLAIM',text='he died',update={'dead':True})
 def test_09_no_production_event_id(self):
  self.assertRaises(Rejected,append,self.a,event_id='EVT-000001',actor='system',kind='SYSTEM_TEST',text='x')
 def test_10_segment_stages_with_head(self):
  x=self.add();s=stage_archive(x);self.assertEqual(s['last_hash'],x['last_hash']);self.assertEqual(s['provocation'],'TEST-P01')
 def test_11_no_empty_seal(self):self.assertRaises(Rejected,stage_archive,self.a)
 def test_12_rotation_with_verified_shard(self):
  x=self.add();s=stage_archive(x);d=archive_digest(s);z=plan_rotation(x,s,'TEST-DOC-001',d);self.assertEqual(z['active_provocation'],'TEST-P02');self.assertEqual(len(z['archives']),1)
 def test_13_refuse_wrong_hash_rotation(self):
  x=self.add();s=stage_archive(x);self.assertRaises(Rejected,plan_rotation,x,s,'TEST-DOC-001','0'*64)
 def test_14_refuse_forged_segment_head(self):
  x=self.add();s=stage_archive(x);s['last_hash']='bad';self.assertRaises(Rejected,plan_rotation,x,s,'TEST-DOC-001',archive_digest(s))
 def test_15_sandbox_namespace(self):
  x=self.a;x['sandbox']=False;self.assertRaises(Rejected,verify,x)
 def test_16_serialization_recovery(self):
  import json
  x=self.add();y=json.loads(json.dumps(x));self.assertTrue(verify(y));self.assertEqual(x,y)
 def test_17_append_player_with_source(self):
  x,status=append(self.a,event_id='TEST-EVT-02',actor='pc',kind='PLAYER_DECLARATION',text='I ask',source_ref='TEST-CHAT-1');self.assertEqual(status,'APPENDED')
 def test_18_two_event_chain(self):
  x=self.add();y,_=append(x,event_id='TEST-EVT-002',actor='system',kind='SYSTEM_TEST',text='Synthetic second');self.assertEqual(y['events'][1]['prev_hash'],y['events'][0]['hash']);self.assertTrue(verify(y))
if __name__=='__main__': unittest.main(verbosity=2)
