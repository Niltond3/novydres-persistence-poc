"""Independent sandbox audit: load FULL archived content, never trust pointer alone.
No Drive access; caller supplies Google Docs readback bytes as dictionaries and checks exact transfer.
Testing only, not a production V5 engine or user-authority guarantee.
"""
from copy import deepcopy
import unittest, json
from pathlib import Path
from webfirst_v31_candidate import digest, validate, valid_event, Violation, prepare_segment
def strict_restore(active, fetch_segment):
    validate(active)  # This alone does NOT load or verify archives.
    state = deepcopy(active['anchor_state'])
    expected_seq = 1
    head = '0' * 64
    seen = set()
    for meta in active['archives']:
        doc_id = meta['doc_id']
        if doc_id not in fetch_segment: raise Violation('Missing archive on Drive')
        segment = fetch_segment[doc_id]
        if segment.get('schema') != 'NOVYDRES_V3_ARCHIVE_STAGED': raise Violation('Archive schema')
        if segment.get('chronicle_id') != active['chronicle_id']: raise Violation('Archive chronicle')
        if digest(segment) != meta['sha256']: raise Violation('Archive digest mismatch')
        for a,b in [('start_seq','start_seq'),('end_seq','end_seq'),('prev_hash','prev_hash'),('head_hash','head_hash'),('provocation','provocation')]:
            if segment.get(a) != meta.get(b): raise Violation('Archive metadata mismatch: '+a)
        if segment['start_seq'] != expected_seq or segment['prev_hash'] != head:
            raise Violation('Global event chain discontinuity')
        if segment['start_state'] != state: raise Violation('Archive start state differs')
        for ev in segment['events']:
            if ev['id'] in seen: raise Violation('Duplicate event across shards')
            state = valid_event(ev, state, head, expected_seq)
            head = ev['hash'];expected_seq += 1;seen.add(ev['id'])
        if expected_seq != segment['end_seq']+1 or head != segment['head_hash']:
            raise Violation('Archive end/head mismatch')
        if state != segment['end_state'] or digest(state) != meta['end_state_hash']:
            raise Violation('Archive state mismatch')
    if active['start_seq'] != expected_seq or active['prev_hash'] != head:
        raise Violation('Active does not extend archive chain')
    if active['archives'] and (active['archive_seam_state'] != state or active['archive_seam_state_hash'] != digest(state)):
        raise Violation('Active seam state mismatch')
    for ev in active['events']:
        if ev['id'] in seen: raise Violation('Duplicate active event')
        state = valid_event(ev, state, head, expected_seq)
        head = ev['hash']; expected_seq += 1;seen.add(ev['id'])
    if active['next_seq'] != expected_seq or active['state'] != state: raise Violation('Active replay mismatch')
    return {'event_count':len(seen),'next_seq':expected_seq,'archives_verified':len(active['archives']), 'head':head, 'final_state':state}
p=Path(__file__).parent/'drive_integration_fixtures'
load=lambda name: json.loads((p/(name+'.txt')).read_text(encoding='ascii'))
ID1='1_dLmz9Wb-F9MqDoNZNmkKTPj7tJWHQlhQ0mMw8pooxg'
ID2='1JSXYBKTDvnGd17UuLCcTJMlzaIiFT8_UvN9-d3LB54g'
class TestExternalArchiveAudit(unittest.TestCase):
 def test_01_active4_replays_archive1_plus_active_events(self):
  r=strict_restore(load('active4'),{ID1:load('archive1')});self.assertEqual((r['archives_verified'],r['event_count'],r['next_seq']),(1,3,4))
 def test_02_future_active5_replays_two_archives_plan(self):
  r=strict_restore(load('active5'),{ID1:load('archive1'),ID2:load('archive2')});self.assertEqual((r['archives_verified'],r['event_count'],r['next_seq']),(2,3,4))
 def test_03_archive_missing_rejected(self):
  with self.assertRaises(Violation): strict_restore(load('active4'),{})
 def test_04_changed_archive_content_rejected(self):
  a=load('archive1');a['events'][0]['text']='Tampered';
  with self.assertRaises(Violation):strict_restore(load('active4'),{ID1:a})
 def test_05_changed_archive_state_rejected(self):
  a=load('archive1');a['end_state']['npc.test.claim']='LIE';
  with self.assertRaises(Violation):strict_restore(load('active4'),{ID1:a})
 def test_06_wrong_archive_identity_rejected(self):
  s=load('active4');s['archives'][0]['doc_id']='OTHER-DOC';
  with self.assertRaises(Violation):strict_restore(s,{ID1:load('archive1')})
 def test_07_wrong_seam_state_rejected(self):
  s=load('active4');s['archive_seam_state']['pc.location']='WRONG';
  with self.assertRaises(Violation):strict_restore(s,{ID1:load('archive1')})
 def test_08_wrong_cross_shard_head_rejected(self):
  s=load('active4');s['prev_hash']='f'*64;
  with self.assertRaises(Violation):strict_restore(s,{ID1:load('archive1')})
 def test_09_stage2_archive_is_not_promoted(self):
  s=load('active4');a=load('archive2'); self.assertEqual(a, prepare_segment(s));self.assertEqual(len(s['archives']),1)
 def test_10_recovery_idempotency_same_event_id(self):
  active=load('active4');self.assertEqual(sum(ev['id']=='S3-EVT-000002' for ev in active['events']),1)
 def test_11_older_event_already_archived(self):
  a=load('active4');self.assertNotIn('S3-EVT-000001',[ev['id'] for ev in a['events']]);self.assertIn('S3-EVT-000001',[ev['id'] for ev in load('archive1')['events']])
 def test_12_end_state_matches_latest_active(self):
  r=strict_restore(load('active4'),{ID1:load('archive1')});self.assertEqual(r['final_state']['pc.location'],'TEST_HALL');self.assertEqual(r['final_state']['xp.spent'],5)
if __name__=='__main__': unittest.main(verbosity=2)
