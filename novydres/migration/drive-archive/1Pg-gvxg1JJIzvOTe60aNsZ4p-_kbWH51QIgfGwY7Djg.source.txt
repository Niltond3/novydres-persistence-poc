"""NOVYDRES Web First V3.1 candidate (staging only).
No network, external service or background execution. Caller must perform Drive CAS + readback.
NEVER import historical narrative as cryptographically committed events.
"""
from __future__ import annotations
from copy import deepcopy
from hashlib import sha256
import json
from typing import Any
SCHEMA = "NOVYDRES_V3_CANDIDATE"
GENESIS = "0" * 64
KINDS = {"PLAYER_DECLARATION", "NPC_ACTION", "WORLD_EVENT", "NPC_CLAIM", "PC_BELIEF", "GM_HYPOTHESIS", "SYSTEM", "CORRECTION"}
NONFACT = {"NPC_CLAIM", "PC_BELIEF", "GM_HYPOTHESIS"}
PROTECTED = {"pc.action", "pc.speech", "pc.location", "xp.spent", "xp.credit_granted", "wp.historical_spent"}
class Violation(ValueError):
    pass
def canon(v: Any) -> str:
    return json.dumps(v, sort_keys=True, ensure_ascii=False, separators=(",", ":"), allow_nan=False)
def digest(v: Any) -> str:
    return sha256(canon(v).encode("utf-8")).hexdigest()
def require(ok: bool, why: str) -> None:
    if not ok: raise Violation(why)
def new_staging(anchor: dict, source: dict) -> dict:
    require(source.get("status") == "DOC_DERIVED_NOT_TECHNICAL_COMMIT", "Provenance needed")
    require(isinstance(source.get("doc_id"), str) and source["doc_id"], "Source doc needed")
    require(isinstance(source.get("revision"), str) and source["revision"], "Source revision needed")
    require(isinstance(anchor, dict), "Anchor must be dict")
    require(anchor.get("xp.total") - anchor.get("xp.spent") == anchor.get("xp.remaining"), "XP inconsistency")
    require(anchor.get("xp.credit_granted") - anchor.get("xp.spent") == anchor.get("xp.credit_available"), "Credit inconsistency")
    return {"schema": SCHEMA, "phase": "STAGING_ONLY", "chronicle_id": "CHR-001",
            "historical_basis": deepcopy(source), "anchor_state": deepcopy(anchor),
            "anchor_hash": digest({"basis":source, "state":anchor}),
            "state": deepcopy(anchor), "active_provocation": "M1.09", "start_seq": 1,
            "next_seq": 1, "prev_hash": GENESIS, "events": [], "archives": [],
            "unmigrated_history": [f"M1.{i:02d}" for i in range(1,10)],
            "rolls_imported": 0}
def apply_deltas(state: dict, deltas: list[dict]) -> dict:
    new = deepcopy(state);seen=set()
    for d in deltas:
        require(isinstance(d,dict) and set(d)=={"field","before","after","reason"}, "Delta shape")
        f=d["field"]
        require(isinstance(f,str) and f not in seen, "Duplicate delta field")
        require(new.get(f)==d["before"] and isinstance(d["reason"],str) and d["reason"],"Before/cause")
        seen.add(f);new[f]=d["after"]
    require(new["xp.total"] - new["xp.spent"] == new["xp.remaining"], "XP budget broken")
    require(new["xp.credit_granted"] - new["xp.spent"] == new["xp.credit_available"], "XP credit broken")
    require(new["xp.spent"] >= 0 and new["xp.remaining"] >= 0 and new["xp.credit_available"] >= 0, "Negative credit")
    return new
def valid_event(e: dict, before: dict, previous: str, sequence: int) -> dict:
    require(e.get("seq") == sequence and e.get("id") == f"S3-EVT-{sequence:06d}", "Global sequence")
    require(e.get("prev_hash") == previous,"Cross-shard predecessor")
    require(e.get("kind") in KINDS and isinstance(e.get("text"),str) and e["text"],"Kind/text")
    require(e.get("actor") and isinstance(e.get("source_ref"),str) and e["source_ref"],"Actor/provenance")
    require(isinstance(e.get("deltas"),list),"Deltas list")
    require(e.get("hash") == digest({k:v for k,v in e.items() if k!="hash"}),"Hash mismatch")
    kind=e["kind"]
    if kind in NONFACT:require(e["deltas"]==[],"Unverified assertion changed facts")
    if e.get("actor")=="pc":require(kind=="PLAYER_DECLARATION","NPC agent took PC action")
    if kind=="PLAYER_DECLARATION":
        require(e.get("actor") == "pc", "PLAYER_DECLARATION must have PC actor")
        require(e.get("verbatim") == e["text"] and e.get("source_ref").startswith("CHAT-"),"Literal declaration required")
    if kind=="CORRECTION":require(e.get("correction_target") and e.get("approval_ref"),"Correction must link and be approved")
    fields={d["field"] for d in e["deltas"]}
    if fields & PROTECTED:
        require(kind=="PLAYER_DECLARATION" and e.get("approval_ref"),"Protected field requires PC's approved literal declaration")
    if "xp.spent" in fields:
        require("xp.remaining" in fields and "xp.credit_available" in fields,"Complete XP debit required")
    if "xp.credit_granted" in fields:require("xp.credit_available" in fields,"Complete credit update required")
    return apply_deltas(before, e["deltas"])
def validate(store: dict) -> dict:
    require(store.get("schema")==SCHEMA and store.get("phase")=="STAGING_ONLY","Not a staging V3")
    require(store.get("chronicle_id")=="CHR-001","Chronicle mismatch")
    require(isinstance(store.get("historical_basis"),dict) and store["historical_basis"].get("status")=="DOC_DERIVED_NOT_TECHNICAL_COMMIT", "Historical provenance degraded")
    require(store.get("anchor_hash")==digest({"basis":store["historical_basis"], "state":store["anchor_state"]}),"Anchor altered")
    require(isinstance(store.get("archives"),list) and isinstance(store.get("events"),list),"Missing ledgers")
    assert_seq=1;hash_head=GENESIS;state=deepcopy(store["anchor_state"])
    for arch in store["archives"]:
        require(arch.get("start_seq")==assert_seq and arch.get("prev_hash")==hash_head,"Archive sequence/head")
        require(arch.get("end_seq")>=assert_seq and arch.get("head_hash") and arch.get("sha256") and arch.get("doc_id"),"Archive manifest incomplete")
        require(arch.get("end_state_hash") and arch.get("verified_readback") is True,"Archive must be read back")
        # The archive itself MUST be reloaded and checked before accepting a new action.
        require(arch.get("validation") == "EXTERNAL_ARCHIVE_REQUIRED", "No archive verification policy")
        assert_seq=arch["end_seq"]+1;hash_head=arch["head_hash"]
    require(store.get("start_seq")==assert_seq and store.get("prev_hash")==hash_head,"Active must extend archives")
    # Replay active after archives requires a verified state at the archive seam.
    if store["archives"]:
        require(store.get("archive_seam_state_hash")==digest(store.get("archive_seam_state")),"Archive seam not provided")
        require(store["archives"][-1]["end_state_hash"]==store["archive_seam_state_hash"],"Archive seam mismatched")
        state=deepcopy(store["archive_seam_state"])
    ids=set();seq=assert_seq;head=hash_head
    for e in store["events"]:
        require(e.get("id") not in ids,"Duplicate ID")
        state=valid_event(e,state,head,seq);head=e["hash"];seq+=1;ids.add(e["id"])
    require(store.get("next_seq")==seq and store.get("state")==state,"Snapshot/sequence not replayable")
    require(store.get("rolls_imported")==0,"Historical rolls must not be fabricated")
    return {"ok":True,"next_seq":seq,"hash_head":head,"active_events":len(store["events"]),"archived_segments":len(store["archives"])}
def append(store: dict, command:dict) -> tuple[dict,str]:
    validate(store)
    expected=command.get("id")
    found=[e for e in store["events"] if e["id"]==expected]
    if found:
        comparable={k:v for k,v in found[0].items() if k not in {"hash","seq","prev_hash"}}
        require(comparable=={k:v for k,v in command.items() if k not in {"hash","seq","prev_hash"}},"Conflicting retry")
        return deepcopy(store),"ALREADY_COMMITTED_IN_ACTIVE"
    require(expected is None or expected == f"S3-EVT-{store['next_seq']:06d}","Never reuse archived/other ID")
    seq=store["next_seq"]
    e=deepcopy(command);e["seq"]=seq;e["id"]=f"S3-EVT-{seq:06d}"
    e["prev_hash"]=store["events"][-1]["hash"] if store["events"] else store["prev_hash"]
    e["hash"]=digest({k:v for k,v in e.items() if k!="hash"})
    state=valid_event(e,store["state"],e["prev_hash"],seq)
    out=deepcopy(store);out["events"].append(e);out["state"]=state;out["next_seq"]+=1
    validate(out);return out,"APPEND_READY_FOR_DRIVE_CAS"
def prepare_segment(store:dict) -> dict:
    validate(store);require(store["events"],"No event to archive")
    return {"schema":"NOVYDRES_V3_ARCHIVE_STAGED","chronicle_id":store["chronicle_id"],
            "provocation":store["active_provocation"],"start_seq":store["start_seq"],
            "end_seq":store["next_seq"]-1,"prev_hash":store["prev_hash"],
            "head_hash":store["events"][-1]["hash"],"events":deepcopy(store["events"]),
            "start_state":deepcopy(store.get("archive_seam_state",store["anchor_state"])),
            "end_state":deepcopy(store["state"])}
def rotate_after_readback(store:dict, segment:dict, doc_id:str, observed:dict, next_provocation:str)->dict:
    validate(store)
    require(observed==segment,"Archive content readback mismatch")
    require(segment==prepare_segment(store),"Archive does not represent active")
    require(next_provocation!=store["active_provocation"],"Next provocation must change")
    archive={"doc_id":doc_id,"provocation":segment["provocation"],"start_seq":segment["start_seq"],
             "end_seq":segment["end_seq"],"prev_hash":segment["prev_hash"],
             "head_hash":segment["head_hash"],"sha256":digest(segment),
             "end_state_hash":digest(segment["end_state"]),"verified_readback":True,
             "validation":"EXTERNAL_ARCHIVE_REQUIRED"}
    new=deepcopy(store);new["archives"].append(archive)
    new["active_provocation"]=next_provocation
    new["events"]=[];new["start_seq"]=store["next_seq"]
    new["prev_hash"]=segment["head_hash"]
    new["archive_seam_state"]=deepcopy(store["state"])
    new["archive_seam_state_hash"]=digest(store["state"])
    validate(new);return new
def plan_cas(old_serialized:str,old_revision:str,new:dict)->dict:
    """Connector-neutral request blueprint, never an automatic Drive write."""
    old=json.loads(old_serialized);validate(old);validate(new)
    require(old_revision,"requiredRevisionId missing")
    require(old["anchor_hash"]==new["anchor_hash"] and old["historical_basis"]==new["historical_basis"],"Anchor/source mutated")
    if len(new["archives"])==len(old["archives"]):
        require(new["archives"]==old["archives"] and new["events"][:len(old["events"])]==old["events"],"Historical prefix changed")
    else:
        require(len(new["archives"])==len(old["archives"])+1 and new["archives"][:len(old["archives"])]==old["archives"],"Invalid rotation")
        last=new["archives"][-1]
        require(bool(old["events"]) and last["start_seq"]==old["start_seq"] and last["end_seq"]==old["next_seq"]-1,
                "Rotation missing active events")
        require(last["prev_hash"]==old["prev_hash"] and last["head_hash"]==old["events"][-1]["hash"],
                "Rotation does not conserve historical head")
        require(last["end_state_hash"]==digest(old["state"]),"Rotation state mismatch")
    return {"write_control":{"requiredRevisionId":old_revision},
            "requests":[{"replaceAllText":{"containsText":{"text":old_serialized,"matchCase":True},
                                           "replaceText":canon(new)}}]}
