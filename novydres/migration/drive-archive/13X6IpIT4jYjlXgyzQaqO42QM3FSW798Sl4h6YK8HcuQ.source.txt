import copy, json, unittest
from webfirst_v31_candidate import *
class V3(unittest.TestCase):
 def setUp(self):
  self.state={'xp.total':519,'xp.spent':44,'xp.remaining':475,'xp.credit_granted':140,'xp.credit_available':96,'pc.location':'house','npc.claim':'UNKNOWN'}
  self.src={'status':'DOC_DERIVED_NOT_TECHNICAL_COMMIT','doc_id':'character-1','revision':'rev1'}
  self.store=new_staging(self.state,self.src)
 def cmd(self,**kw):
  d={'kind':'NPC_CLAIM','actor':'npc-1','text':'Says they will meet','source_ref':'DOC-M1.09','deltas':[]}
  d.update(kw);return d
 def with_event(self,**kw):return append(self.store,self.cmd(**kw))[0]
 def test_01_valid(self):self.assertTrue(validate(self.store)['ok'])
 def test_02_provenance_source_required(self):
  with self.assertRaises(Violation):new_staging(self.state,{'status':'COMMITTED','doc_id':'x','revision':'x'})
 def test_03_xp_initial_bad(self):
  s=copy.deepcopy(self.state);s['xp.remaining']=999
  with self.assertRaises(Violation):new_staging(s,self.src)
 def test_04_claim_no_mutation(self):
  with self.assertRaises(Violation):self.with_event(deltas=[{'field':'npc.claim','before':'UNKNOWN','after':'TRUE','reason':'claim'}])
 def test_05_valid_claim(self):self.assertEqual(self.with_event()['next_seq'],2)
 def test_06_replay_and_chain(self):
  a=self.with_event();b,_=append(a,self.cmd(text='Another claim'))
  self.assertEqual(b['events'][1]['prev_hash'],a['events'][0]['hash']);self.assertTrue(validate(b)['ok'])
 def test_07_tampered_text(self):
  s=self.with_event();s['events'][0]['text']='tampered'
  with self.assertRaises(Violation):validate(s)
 def test_08_tampered_snapshot(self):
  s=self.with_event();s['state']['pc.location']='wrong'
  with self.assertRaises(Violation):validate(s)
 def test_09_false_pc_agency(self):
  with self.assertRaises(Violation):self.with_event(kind='NPC_ACTION',actor='pc')
 def test_10_pc_without_quote(self):
  with self.assertRaises(Violation):self.with_event(kind='PLAYER_DECLARATION',actor='pc',source_ref='CHAT-11')
 def test_11_pc_protected_field_requires_approval(self):
  with self.assertRaises(Violation):self.with_event(kind='PLAYER_DECLARATION',actor='pc',text='Enter',verbatim='Enter',source_ref='CHAT-11',deltas=[{'field':'pc.location','before':'house','after':'street','reason':'declared'}])
 def test_12_pc_protected_field_approved(self):
  r=self.with_event(kind='PLAYER_DECLARATION',actor='pc',text='Enter',verbatim='Enter',source_ref='CHAT-11',approval_ref='CHAT-11',deltas=[{'field':'pc.location','before':'house','after':'street','reason':'declared'}]);self.assertEqual(r['state']['pc.location'],'street')
 def test_13_negative_spend(self):
  with self.assertRaises(Violation):self.with_event(kind='PLAYER_DECLARATION',actor='pc',text='Buy',verbatim='Buy',source_ref='CHAT-11',approval_ref='CHAT-11',deltas=[{'field':'xp.spent','before':44,'after':50,'reason':'purchase'}])
 def test_14_xp_complete_spend(self):
  ds=[{'field':'xp.spent','before':44,'after':50,'reason':'purchase'}, {'field':'xp.remaining','before':475,'after':469,'reason':'purchase'},{'field':'xp.credit_available','before':96,'after':90,'reason':'purchase'}]
  a=self.with_event(kind='PLAYER_DECLARATION',actor='pc',text='Buy',verbatim='Buy',source_ref='CHAT-11',approval_ref='CHAT-11',deltas=ds);self.assertEqual(a['state']['xp.credit_available'],90)
 def test_15_wrong_before(self):
  with self.assertRaises(Violation):self.with_event(kind='PLAYER_DECLARATION',actor='pc',text='Enter',verbatim='Enter',source_ref='CHAT-11',approval_ref='CHAT-11',deltas=[{'field':'pc.location','before':'wrong','after':'street','reason':'declared'}])
 def test_16_idempotent(self):
  a=self.with_event();b,status=append(a,self.cmd(id='S3-EVT-000001'))
  self.assertEqual(status,'ALREADY_COMMITTED_IN_ACTIVE');self.assertEqual(b,a)
 def test_17_conflicting_retry(self):
  a=self.with_event()
  with self.assertRaises(Violation):append(a,self.cmd(id='S3-EVT-000001',text='different'))
 def test_18_archive_replay_seam(self):
  a=self.with_event();s=prepare_segment(a);b=rotate_after_readback(a,s,'doc-test',copy.deepcopy(s),'M1.10')
  self.assertEqual(b['start_seq'],2);self.assertEqual(b['prev_hash'],a['events'][0]['hash']);self.assertTrue(validate(b)['ok'])
 def test_19_archive_wrong_readback(self):
  a=self.with_event();s=prepare_segment(a);s2=copy.deepcopy(s);s2['events'][0]['text']='changed'
  with self.assertRaises(Violation):rotate_after_readback(a,s,'doc-test',s2,'M1.10')
 def test_20_archive_continuous_global_ids(self):
  a=self.with_event();s=prepare_segment(a);b=rotate_after_readback(a,s,'doc-test',copy.deepcopy(s),'M1.10');c,_=append(b,self.cmd(text='Continued'))
  self.assertEqual(c['events'][0]['id'],'S3-EVT-000002');self.assertEqual(c['events'][0]['prev_hash'],s['head_hash'])
 def test_21_correction_authorization(self):
  with self.assertRaises(Violation):self.with_event(kind='CORRECTION',text='corrected')
 def test_22_anchor_tampered(self):
  a=self.store;a['anchor_state']['xp.remaining']=1
  with self.assertRaises(Violation):validate(a)
 def test_23_rolls_must_not_be_imported(self):
  a=self.store;a['rolls_imported']=1
  with self.assertRaises(Violation):validate(a)
 def test_24_no_archiving_empty(self):
  with self.assertRaises(Violation):prepare_segment(self.store)
 def test_25_plan_cas_prefix_protection(self):
  a=self.with_event();b,_=append(a,self.cmd(text='Another'))
  x=plan_cas(canon(a),'rev1',b);self.assertEqual(x['write_control']['requiredRevisionId'],'rev1')
 def test_26_plan_cas_tampering(self):
  a=self.with_event();b,_=append(a,self.cmd(text='Another'));b['events'][0]['text']='oops'
  with self.assertRaises(Violation):plan_cas(canon(a),'rev1',b)
 def test_27_no_unsupported_meta_source(self):
  a=self.with_event();a['historical_basis']['revision']='changed'
  with self.assertRaises(Violation):validate(a)
 def test_28_archive_pointer_integrity(self):
  a=self.with_event();s=prepare_segment(a);b=rotate_after_readback(a,s,'doc-test',copy.deepcopy(s),'M1.10');b['archives'][0]['end_state_hash']='tamper'
  with self.assertRaises(Violation):validate(b)
 def test_29_nonfact_belief(self):
  with self.assertRaises(Violation):self.with_event(kind='PC_BELIEF',deltas=[{'field':'npc.claim','before':'UNKNOWN','after':'TRUE','reason':'belief'}])
 def test_30_serialize_roundtrip(self):
  a=self.with_event();r=json.loads(canon(a));self.assertEqual(a,r);self.assertTrue(validate(r)['ok'])
 def test_31_reject_npc_forging_pc_declaration(self):
  with self.assertRaises(Violation):self.with_event(kind='PLAYER_DECLARATION',actor='npc-1',text='I move',verbatim='I move',source_ref='CHAT-FORGED',approval_ref='CHAT-FORGED',deltas=[{'field':'pc.location','before':'house','after':'street','reason':'forged'}])
 def test_32_approve_real_pc_declaration(self):
  a=self.with_event(kind='PLAYER_DECLARATION',actor='pc',text='I move',verbatim='I move',source_ref='CHAT-OK',approval_ref='CHAT-OK',deltas=[{'field':'pc.location','before':'house','after':'street','reason':'declared'}]);self.assertEqual(a['state']['pc.location'],'street')
 def test_33_npc_action_protected_player_speech_rejected(self):
  with self.assertRaises(Violation):self.with_event(kind='NPC_ACTION',actor='npc-1',deltas=[{'field':'pc.speech','before':None,'after':'Sure','reason':'NPC_fabrication'}])
if __name__=='__main__':unittest.main(verbosity=2)
