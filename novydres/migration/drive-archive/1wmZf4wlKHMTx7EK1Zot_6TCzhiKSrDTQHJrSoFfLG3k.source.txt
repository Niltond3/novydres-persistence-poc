def fixture(name, id):
    return RemoteDoc(id,'TEST-REV-'+name,(F/(name+'.txt')).read_text(encoding='utf-8').strip())
def archives(one=True,two=False):
    d={};
    if one:d[ID1]=fixture('archive1',ID1)
    if two:d[ID2]=fixture('archive2',ID2)
    return d
class TestIOGate(unittest.TestCase):
    def setUp(self):
        self.a4=fixture('active4',ACT)
        self.a5=fixture('active5',ACT)
        self.a6=fixture('active6',ACT)
    def base_command(self):
        s=self.a5.parsed();return {'id':'S3-EVT-000004','kind':'NPC_CLAIM','actor':'TEST-NPC-02',
          'text':'Synthetic claim only: the gate is locked','source_ref':'TEST-EVIDENCE-004','deltas':[]}
    def test_01_archive1_loaded(self):self.assertEqual(readback_gate(self.a4,archives())['event_count'],3)
    def test_02_archives2_loaded(self):self.assertEqual(readback_gate(self.a5,archives(two=True))['event_count'],3)
    def test_03_archives2_activeevent(self):self.assertEqual(readback_gate(self.a6,archives(two=True))['event_count'],4)
    def test_04_no_segment_fails(self):
        with self.assertRaises(Violation):readback_gate(self.a6,{})
    def test_05_missing_archive2_fails(self):
        with self.assertRaises(Violation):readback_gate(self.a6,archives())
    def test_06_wrong_id_fails(self):
        a=archives(two=True);a[ID2]=RemoteDoc('FORGED-ID','r',a[ID2].text)
        with self.assertRaises(Violation):readback_gate(self.a6,a)
    def test_07_invalid_json_fails(self):
        with self.assertRaises(Violation):readback_gate(RemoteDoc(ACT,'rev','{'),archives(two=True))
    def test_08_tampered_archived_content_fails(self):
        a=archives(two=True);p=a[ID2].parsed();p['events'][0]['text']='LIE';a[ID2]=RemoteDoc(ID2,'rev',canon(p))
        with self.assertRaises(Violation):readback_gate(self.a6,a)
    def test_09_tampered_shard_meta_fails(self):
        p=self.a6.parsed();p['archives'][1]['sha256']='0'*64
        with self.assertRaises(Violation):readback_gate(RemoteDoc(ACT,'rev',canon(p)),archives(two=True))
    def test_10_tampered_active_state_fails(self):
        p=self.a6.parsed();p['state']['pc.location']='NO'
        with self.assertRaises(Violation):readback_gate(RemoteDoc(ACT,'rev',canon(p)),archives(two=True))
    def test_11_existing_active_event_found(self):
        e=self.a6.parsed()['events'][0]
        p=prepare_append_from_readbacks(self.a6,archives(two=True),{k:v for k,v in e.items() if k not in ('hash','prev_hash','seq')})
        self.assertEqual(p['status'],'ALREADY_COMMITTED_READBACK')
    def test_12_existing_archive_event_found(self):
        e=archives()[ID1].parsed()['events'][0]
        p=prepare_append_from_readbacks(self.a6,archives(two=True),{k:v for k,v in e.items() if k not in ('hash','prev_hash','seq')})
        self.assertEqual(p['status'],'ALREADY_COMMITTED_READBACK')
    def test_13_conflicting_retry_rejected(self):
        e=self.a6.parsed()['events'][0];command={k:v for k,v in e.items() if k not in ('hash','prev_hash','seq')};command['text']='FORGED'
        with self.assertRaises(Violation):prepare_append_from_readbacks(self.a6,archives(two=True),command)
    def test_14_new_event_plan(self):
        command={'id':'S3-EVT-000005','kind':'NPC_CLAIM','actor':'npc-claimant','text':'Further test claim','source_ref':'DOC-TEST-005','deltas':[]}
        result=prepare_append_from_readbacks(self.a6,archives(two=True),command)
        self.assertEqual(result['status'],'READY_CAS');self.assertEqual(result['write']['write_control']['requiredRevisionId'],self.a6.revision)
        self.assertEqual(len(json.loads(result['expected_text'])['events']),2)
    def test_15_append_claim_delta_blocked(self):
        command={'id':'S3-EVT-000005','kind':'NPC_CLAIM','actor':'npc','text':'Claim','source_ref':'DOC-5',
                 'deltas':[{'field':'pc.location','before':'TEST_HALL','after':'SOMEWHERE','reason':'LIE'}]}

    def test_16_plan_fails_if_no_readback(self):

    def test_17_ack_with_readback_exact(self):

    def test_18_lost_ack_readback_exact(self):

        self.assertEqual(evaluate_ack(plan,self.a6,archives(two=True))['status'],'ACK_LOST_BUT_READBACK_VERIFIED')
    def test_19_failed_ack_remains_pending(self):

    def test_20_bad_occurrence_fails(self):
