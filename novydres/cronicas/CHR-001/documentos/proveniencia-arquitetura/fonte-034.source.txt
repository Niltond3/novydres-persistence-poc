#!/usr/bin/env python3
import hashlib,json,re,sys
from pathlib import Path
ROOT=Path(sys.argv[1] if len(sys.argv)>1 else '.')
checks=[]
def check(name,condition):
    if not condition: raise SystemExit('D06_SOURCE_V003_BLOCK: '+name)
    checks.append(name)
def read(p): return (ROOT/p).read_text(encoding='utf-8')
def blob(p):
    b=(ROOT/p).read_bytes(); return hashlib.sha1(b'blob '+str(len(b)).encode()+b'\0'+b).hexdigest()
def ids(s): return re.findall(r'CHR\d{3}-[A-Z0-9][A-Z0-9-]*-\d{6,}-\d{3}',s)
for p in ['novydres/chronicles/CHR-001/episodes/M2.03_v018.source.txt','novydres/chronicles/CHR-001/episodes/M2.04_v018.source.txt','novydres/chronicles/CHR-001/episodes/M3.01_v006.source.txt']:
    s=read(p); check(p+' has 1970 approval', 'APPROVAL-D06-002' in s and '1970' in s)
    check(p+' has no stale 1970s calendar anchor',not re.search(r'fim\s+(?:dos\s+)?anos\s+1970|fim\s+da\s+d[eé]cada\s+(?:de\s+)?1970|late\s*1970s|final\s+1970s|mortal\s+1970s',s,re.I))
    prior={'M2.03':'novydres/chronicles/CHR-001/episodes/M2.03_v017.source.txt','M2.04':'novydres/chronicles/CHR-001/episodes/M2.04_v017.source.txt','M3.01':'novydres/chronicles/CHR-001/episodes/M3.01_v005.source.txt'}[p.split('/')[-1].split('_')[0]]
    check(p+' event IDs unchanged',ids(s)==ids(read(prior)))
m3=read('novydres/chronicles/CHR-001/episodes/M3.01_v006.source.txt')
check('M3.01 is in 1970','1970' in m3 and 'late1970s' not in m3)
check('M3.01 preserves mortality','mortal' in m3.lower())
check('M3.01 has no recorded Embrace',not re.search(r'(?:Abraço|Embrace)\s+(?:ocorreu|occurred)',m3,re.I))
check('C03.15 anomaly retained','C03.15' in m3 and ('1970s' in m3[m3.index('C03.15'):m3.index('C03.15')+1800] or 'anos 1970' in m3[m3.index('C03.15'):m3.index('C03.15')+1800]))
state=read('novydres/chronicles/CHR-001/character/state_v044.source.txt')
check('state lineage says v043 compatible','The original state_v043 is cumulative' in state)
check('state cumulative chronology has no stale late-1970s labels',not re.search(r'fim\s+(?:dos\s+)?anos\s+1970|fim\s+da\s+d[eé]cada\s+(?:de\s+)?1970|mortal\s+1970s',state,re.I))
check('state records chronology label reconciliation','cumulative chronology labels in this new state version have been reconciled to the player-authorized year 1970' in state)
check('state retracts false mismatch','RETRACTED' in state and 'provenance-incompatible' in state)
check('state retains C05/G28','STATE v043 — CHR-001 / M3.01 C05 CLOSED / G28' in state)
check('state keeps C06 unstarted','C06 NOT_STARTED' in state)
check('state mortal and no Embrace','MORTAL' in state and 'Embrace NOT_OCCURRED' in state)
check('state cites future deadline','1970-12-25' in state and 'unoccurred' in state.lower())
variant=json.loads(read('novydres/runtime/narrative-architecture/program/deliverables/D12_PLAYER_OPEN_VARIANT_V001.json'))
check('variant limited planning scope',variant['status']=='APPROVED_LIMITED_SCOPE' and variant['authorized_scope']=='Dramaturge continuity planning for CHR-001 only')
check('variant blocks hidden details','Do not inspect' in ' '.join(variant['restrictions']))
live=json.loads(read('novydres/chronicles/CHR-001/LIVE_v038.json'))
check('LIVE C05 current',live['scene']['last_event_id']=='CHR001-M301-C05-22-20261109-001')
check('LIVE C06 not started','C06' in live['stage']['M3_to_M5'] or live['scene']['status']=='CLOSED')
check('LIVE is mortal and 1970',live['stage']['embrace_occurred'] is False and '1970' in live['stage']['diegetic_period'])
index=json.loads(read('novydres/chronicles/CHR-001/HISTORY_INDEX_v038.json'))
check('index points to M3.01 v006',index['entrypoints']['current_episode'].endswith('M3.01_v006.source.txt'))
check('index points to state v044',index['entrypoints']['current_character_state'].endswith('state_v044.source.txt'))
check('index current event remains C05',index['current_event']['event_id']=='CHR001-M301-C05-22-20261109-001')
cycle=json.loads(read('novydres/chronicles/CHR-001/orchestration/CYCLE_V0034.json'))
check('cycle transition SOURCE_RECONCILED',cycle['transition']=='SOURCE_RECONCILED')
check('cycle role DRAMATURGO',cycle['next_role']=='DRAMATURGO' and cycle['checkpoint']['next_scene_status']=='NOT_STARTED')
check('cycle last event unchanged',cycle['checkpoint']['last_event_id']=='CHR001-M301-C05-22-20261109-001')
manifest=json.loads(read('novydres/MANIFEST_V48.json'))
check('manifest points to CYCLE_V0034',manifest['runtime']['orchestration']['current_cycle'].endswith('CYCLE_V0034.json'))
checksums={p:blob(p) for p in ['novydres/chronicles/CHR-001/episodes/M2.03_v018.source.txt','novydres/chronicles/CHR-001/episodes/M2.04_v018.source.txt','novydres/chronicles/CHR-001/episodes/M3.01_v006.source.txt','novydres/chronicles/CHR-001/character/state_v044.source.txt']}
print(json.dumps({'validator':'D06_SOURCE_VALIDATE_V003.py','version':'V003','status':'PASS','checks_passed':len(checks),'checks_total':len(checks),'checks':checks,'input_blobs':checksums},ensure_ascii=False,indent=2))
